# --- Smart Learning Error Log System (Render/Linux এর জন্য পাথ পরিবর্তন করা হয়েছে) ---
ERROR_LOG_DIR = "./logs"
ERROR_LOG_FILE = os.path.join(ERROR_LOG_DIR, "error_log.txt")
SCREENSHOT_DIR = os.path.join(ERROR_LOG_DIR, "screenshots")
USED_CATS_FILE = os.path.join(ERROR_LOG_DIR, "used_categories.txt")

def log_error(driver, context_msg, exception_msg):
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    url = "N/A"
    if driver:
        try:
            url = driver.current_url
        except:
            pass
            
    log_entry = f"[{timestamp}] Context: {context_msg} | Error: {exception_msg} | URL: {url}\n"
    
    try:
        with open(ERROR_LOG_FILE, "a", encoding="utf-8") as f:
            f.write(log_entry)
    except Exception as e:
        print(f"Log file write error: {e}")
        
    ss_path = None
    if driver:
        try:
            ss_name = f"error_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png"
            ss_path = os.path.join(SCREENSHOT_DIR, ss_name)
            driver.save_screenshot(ss_path)
        except:
            pass
            
    return ss_path

def cleanup_driver(driver):
    if driver:
        temp_dir = getattr(driver, 'temp_profile_dir', None)
        try:
            driver.quit()
        except:
            pass
        if temp_dir and os.path.exists(temp_dir):
            try:
                shutil.rmtree(temp_dir, ignore_errors=True)
            except:
                pass

def perform_follow_action(driver, chat_id):
    try:
        bot.send_message(chat_id, "🔍 'Suggested for you' লেখা খোঁজা হচ্ছে...")
        suggested_found = False
        for _ in range(3):
            driver.execute_script("window.scrollBy(0, 300);")
            try:
                WebDriverWait(driver, 2).until(EC.presence_of_element_located((By.XPATH, "//*[contains(text(), 'Suggested for you')]")))
                suggested_found = True
                break
            except:
                continue
                
        if not suggested_found:
            bot.send_message(chat_id, "ℹ️ 'Suggested for you' লেখা পাওয়া যায়নি, ফলো স্টেপ স্কিপ করা হলো।")
            return
            
        target_follows = random.randint(1, 6)
        followed_count = 0
        
        while followed_count < target_follows:
            follow_elements = driver.find_elements(By.XPATH, "//*[normalize-space(text())='Follow' and (self::button or self::a or self::div[@role='button'] or self::span)]")
            valid_btn = None
            for el in follow_elements:
                if el.is_displayed():
                    valid_btn = el
                    break
                
            if not valid_btn:
                driver.execute_script("window.scrollBy(0, 500);")
                time.sleep(2)
                continue
                    
            btn = valid_btn
            driver.execute_script("arguments[0].scrollIntoView({block: 'center'});", btn)
            time.sleep(random.uniform(1.0, 2.5))
            driver.execute_script("arguments[0].click();", btn)
            
            followed_count += 1
            bot.send_message(chat_id, f"➕ ফলো করা হয়েছে ({followed_count}/{target_follows})।")
            time.sleep(random.uniform(8.0, 15.0))
                
        bot.send_message(chat_id, f"👥 সর্বমোট {followed_count} টি একাউন্টে ফলো করা সম্পন্ন হয়েছে।")
            
    except Exception as e:
        log_error(driver, "Follow Action", str(e))

            actions.key_down(Keys.CONTROL).send_keys('a').key_up(Keys.CONTROL).perform()
            actions.send_keys(Keys.DELETE).perform()
            actions.send_keys(code_to_enter).perform()
            
            actions.send_keys(Keys.TAB).perform()
            actions.send_keys(Keys.TAB).perform()
            actions.send_keys(Keys.TAB).perform()
            actions.send_keys(Keys.ENTER).perform()
            
            try:
                WebDriverWait(driver, 15).until(lambda d: "auth_platform/codeentry" not in d.current_url or "incorrect" in d.page_source.lower() or "accounts/suspended" in d.current_url)
                new_url = driver.current_url
                new_source = driver.page_source.lower()
                
                if "accounts/suspended" in new_url:
                    blacklist.add(username.lower())
                    with open("blacklist.txt", "a") as f: f.write(f"{username.lower()}\n")
                    cleanup_driver(driver)
                    active_driver = None
                    browser_lock.release()
                    return
                
                if "auth_platform/codeentry" in new_url or "incorrect" in new_source:
                    bot.send_message(chat_id, "❌ কোড ভুল! আবার পাঠান:")
                    continue
                else:
                    bot.send_message(chat_id, "✅ কোড সফল হয়েছে!")
                    check_home_page(driver, username, chat_id, email_to_invite)
                    return
            except:
                bot.send_message(chat_id, "⚠️ কোড দেওয়ার পর রেসপন্স আসছে না।")
                return
                
    except Exception as e:
        log_error(driver, "Code Entry Process", str(e))
        cleanup_driver(driver)
        active_driver = None
        if browser_lock.locked(): browser_lock.release()


@bot.message_handler(commands=['stop'])
def handle_stop(message):
    global active_driver, waiting_for_code
    if message.chat.id != YOUR_CHAT_ID: return
        
    if active_driver:
        if waiting_for_code:
            waiting_for_code = False
            code_event.set()
            
        bot.send_message(message.chat.id, "🍪 ব্রাউজার বন্ধ করা হচ্ছে...")
        try:
            cleanup_driver(active_driver)
            active_driver = None
            browser_lock.release()
            bot.send_message(message.chat.id, "🛑 ব্রাউজার বন্ধ হয়েছে।")
        except Exception as e:
            bot.send_message(message.chat.id, f"বন্ধ করার সময় সমস্যা: {e}")
    else:
        bot.send_message(message.chat.id, "বর্তমানে কোনো ব্রাউজার চলছে না।")


@bot.message_handler(func=lambda message: True)
def handle_message(message):
    global waiting_for_code, received_code
    if message.chat.id != YOUR_CHAT_ID: return
        
    text = message.text.strip()
    if text.startswith('/'): return
        
    if waiting_for_code:
        received_code = text
        code_event.set()
        bot.reply_to(message, "📩 কোড গ্রহণ করা হয়েছে, বসানো হচ্ছে...")
        return
        
    parts = text.split()
    if len(parts) < 2:
        bot.reply_to(message, "⚠️ ইউজারনেম এবং ইনভাইট করার মেইল একসাথে দিন। যেমন: username invite@email.com")
        return

    username = parts[0]
    email_to_invite = parts[1]

    if username.lower() in blacklist:
        bot.reply_to(message, "🚫 এই ইউজারনেম ব্ল্যাকলিস্ট করা আছে।")
        return

    bot.reply_to(message, f"ইউজারনেম: {username}\nইনভাইট মেইল: {email_to_invite}\nChrome ব্রাউজারে লগিন করা হচ্ছে...")
    thread = threading.Thread(target=login_to_instagram, args=(username, message.chat.id, email_to_invite))
    thread.start()
